cmake_minimum_required(VERSION 3.20)
project(VulkanTriangle LANGUAGES CXX)

# Включаем стандарт C++20
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_CXX_USE_RESPONSE_FILE_FOR_INCLUDES OFF)
set(CMAKE_C_USE_RESPONSE_FILE_FOR_INCLUDES OFF)

# 1. Находим Vulkan SDK (CMake найдет его сам через переменные окружения)
find_package(Vulkan REQUIRED)

# 2. Определяем пути к зависимостям
set(IMGUI_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/imgui")
set(GLFW_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/glfw")

# Находим точный путь к glfw3.lib (так CMake не промахнется с путями)
find_library(GLFW_LIBRARY 
    NAMES glfw3
    PATHS "${GLFW_DIR}/lib-vc2022"
    REQUIRED
)

# 3. Собираем ImGui как статическую библиотеку
set(IMGUI_SOURCES
    ${IMGUI_DIR}/imgui.cpp
    ${IMGUI_DIR}/imgui_demo.cpp
    ${IMGUI_DIR}/imgui_draw.cpp
    ${IMGUI_DIR}/imgui_tables.cpp
    ${IMGUI_DIR}/imgui_widgets.cpp
    ${IMGUI_DIR}/backends/imgui_impl_glfw.cpp
    ${IMGUI_DIR}/backends/imgui_impl_vulkan.cpp
)
add_library(imgui STATIC ${IMGUI_SOURCES})

target_include_directories(imgui PUBLIC 
    ${IMGUI_DIR}
    ${IMGUI_DIR}/backends
)
target_include_directories(imgui PRIVATE 
    ${Vulkan_INCLUDE_DIRS}
    ${GLFW_DIR}/include
)
target_link_libraries(imgui PRIVATE 
    Vulkan::Vulkan
    ${GLFW_LIBRARY}
)

# 4. Создаем твой исполняемый файл
add_executable(${PROJECT_NAME} 
    src/main.cpp
    src/VulkanManager/VulkanManager.cpp
    src/Core/ImGuiLayer/ImGuiLayer.cpp
    src/Core/window.cpp
    src/ECS/TransformComponents.cpp
    src/VkInitModules/VkPipelineModule/VkPipelineModule.cpp

    src/VkInitModules/VkPipelineModule/VkPipelineModule.cpp
    src/VkInitModules/VkInstanceModule/VkInstanceModule.cpp
    src/VkInitModules/VkDeviceModule/VkDeviceModule.cpp
    src/VkInitModules/VkRenderPassModule/VkRenderPassModule.cpp
    src/VkInitModules/VkSurfaceModule/VkSurfaceModule.cpp
    src/VkInitModules/VkSwapchainModule/VkSwapchainModule.cpp
)

# Указываем, что используем статическую версию GLFW
target_compile_definitions(${PROJECT_NAME} PRIVATE GLFW_STATIC)

# Настройки инклудов для основного проекта
target_include_directories(${PROJECT_NAME} PRIVATE
    ${Vulkan_INCLUDE_DIRS}
    ${GLFW_DIR}/include
    ${CMAKE_CURRENT_SOURCE_DIR}/src
)

# Линкуем библиотеки (используем переменную ${GLFW_LIBRARY} вместо простого текста "glfw3")
target_link_libraries(${PROJECT_NAME} PRIVATE 
    imgui
    Vulkan::Vulkan
    ${GLFW_LIBRARY}
    gdi32          # Системные библиотеки Windows
    user32 
    kernel32
    shell32        
    imm32          
)
